SDK Usage — Code

import { SlytherinDexClient } from "slytherin-dex-sdk";

const dex = new SlytherinDexClient({
  baseUrl: "https://api.slytherin.io",
});

// Fetch pool reserves
const reserves = await dex.getReserves();

// Get a swap quote (50 bps = 0.5 %)
const quote = await dex.getQuote("algo_to_b", 1_000_000, 50);

// Build unsigned swap txn group
